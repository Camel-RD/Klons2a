ALTER PROCEDURE SP_M_DEL_FINDOC (
    PIDDOCM integer)
AS
DECLARE VARIABLE VDOCID INT;
BEGIN
    for select f_opsd.id
    from f_opsd
    where (f_opsd.iddocm is not null) and (f_opsd.iddocm = :PIDDOCM)
    into :VDOCID
    do
    begin
        delete from f_ops where f_ops.docid = :VDOCID;
    end
    delete from f_opsd
    where (f_opsd.iddocm is not null) and (f_opsd.iddocm = :PIDDOCM);
END
^
