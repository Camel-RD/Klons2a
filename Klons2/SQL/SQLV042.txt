  START WITH 1

    ID INTEGER NOT NULL,
    NAME VARCHAR(50) DEFAULT '?' NOT NULL,
    BANK VARCHAR(8) NOT NULL,
    ACCOUNT VARCHAR(21) DEFAULT '?' NOT NULL);

CREATE TABLE F_PMT_MSG (
    ID INTEGER NOT NULL,
    MSGID INTEGER NOT NULL,
    MSGIDSTR VARCHAR(15) NOT NULL,
    DT TIMESTAMP DEFAULT LOCALTIMESTAMP NOT NULL,
    ACCOUNT INTEGER NOT NULL,
    CT INTEGER DEFAULT 0 NOT NULL,
    TP INTEGER DEFAULT 0 NOT NULL,
    AMOUNT MONEY DEFAULT 0 NOT NULL,
    DESCR VARCHAR(200));

CREATE TABLE F_PMT_TRFTRX (
    ID INTEGER NOT NULL,
    IDMSG INTEGER NOT NULL,
    ID1 INTEGER DEFAULT 0 NOT NULL,
    ID2 INTEGER DEFAULT 0 NOT NULL,
    ID1STR VARCHAR(15),
    ID2STR VARCHAR(15),
    CLID VARCHAR(15) NOT NULL,
    AMOUNT MONEY DEFAULT 0 NOT NULL,
    DETAILS VARCHAR(70) NOT NULL);


ALTER TABLE F_PMT_ACCOUNTS ADD CONSTRAINT PK_F_PMT_ACCOUNTS PRIMARY KEY (ID);

ALTER TABLE F_PMT_MSG ADD CONSTRAINT PK_F_PMT_MSG PRIMARY KEY (ID);

CREATE PROCEDURE SP_F_PMT_TRFTRX_SEL(
  PIDMSG INTEGER)
RETURNS (
  ID INTEGER,
  IDMSG INTEGER,
  ID1 INTEGER,
  ID2 INTEGER,
  ID1STR VARCHAR(15),
  ID2STR VARCHAR(15),
  CLID VARCHAR(15),
  AMOUNT NUMERIC(15,2),
  DETAILS VARCHAR(70))
AS
begin
  for select id,
             idmsg,
             id1,
             id2,
             id1str,
             id2str,
             clid,
             amount,
             details
      from f_pmt_trftrx
      where idmsg = :PIDMSG
      into :id,
           :idmsg,
           :id1,
           :id2,
           :id1str,
           :id2str,
           :clid,
           :amount,
           :details
  do
  begin
    suspend;
  end
end
^
